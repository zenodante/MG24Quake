cmake_minimum_required(VERSION 3.20)
project(quake_mac C)
set(CMAKE_C_STANDARD 11)
find_package(SDL2 CONFIG REQUIRED)
find_package(Threads REQUIRED)
option(QMAC_SANITIZE "Enable undefined behavior sanitizer" ON)
set(SHARED ../rp2350)
add_executable(quake_mac main.c service_sdl.c
    ${SHARED}/qpak.c ${SHARED}/qbsp.c ${SHARED}/qrender.c
    ${SHARED}/qcollision.c ${SHARED}/qmix.c)
target_include_directories(quake_mac PRIVATE . ${SHARED})
target_compile_definitions(quake_mac PRIVATE QR_XIP_ONLY=1)
target_compile_options(quake_mac PRIVATE -Wall -Wextra -Werror -O2 -g -fno-omit-frame-pointer -ffp-contract=off)
target_link_libraries(quake_mac PRIVATE SDL2::SDL2 Threads::Threads)
if(QMAC_SANITIZE)
    target_compile_options(quake_mac PRIVATE -fsanitize=undefined -fno-sanitize-recover=all)
    target_link_options(quake_mac PRIVATE -fsanitize=undefined)
endif()
enable_testing()
option(QMAC_MG24_RENDERER "Use original MG24 edge/surface/span renderer" ON)
if(QMAC_MG24_RENDERER)
    set(MG24 ../../QuakeMG24/Quake)
    set(MG24_SOURCES r_main r_misc r_bsp r_draw r_edge r_surf r_light r_efrag
        d_edge d_fill d_init d_modech d_scan d_sky d_vars mathlib tabmath cvar)
    list(TRANSFORM MG24_SOURCES PREPEND "${MG24}/")
    list(TRANSFORM MG24_SOURCES APPEND ".c")
    add_library(mg24_renderer STATIC mg24_renderer.c ${MG24_SOURCES})
    target_include_directories(mg24_renderer PRIVATE . ${SHARED} ${MG24} ../../QuakeMG24/src)
    target_compile_options(mg24_renderer PRIVATE -include ${CMAKE_CURRENT_SOURCE_DIR}/mg24_config.h
        -Wno-everything -O2 -g -ffp-contract=off -fno-omit-frame-pointer)
    if(QMAC_SANITIZE)
        target_compile_options(mg24_renderer PRIVATE -fsanitize=undefined -fno-sanitize-recover=all)
        target_link_options(quake_mac PRIVATE -fsanitize=undefined)
    endif()
    target_link_libraries(mg24_renderer PRIVATE SDL2::SDL2)
    add_executable(qnative_pack native_pack_main.c ${SHARED}/qpak.c ${SHARED}/qbsp.c)
    target_include_directories(qnative_pack PRIVATE . ${SHARED})
    target_link_libraries(qnative_pack PRIVATE mg24_renderer)
    target_link_options(qnative_pack PRIVATE -Wl,-dead_strip)
    if(QMAC_SANITIZE)
        target_link_options(qnative_pack PRIVATE -fsanitize=undefined)
    endif()
    target_link_libraries(quake_mac PRIVATE mg24_renderer)
    target_compile_definitions(quake_mac PRIVATE QMAC_MG24_RENDERER=1)
    target_link_options(quake_mac PRIVATE -Wl,-dead_strip -Wl,-map,${CMAKE_CURRENT_BINARY_DIR}/quake_mac.map)
endif()
